cmake_minimum_required(VERSION 3.20)
project(LogScan VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(LOGSCAN_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(logscan_core src/parser.c src/stats.c src/report.c)
target_include_directories(logscan_core PUBLIC include)
add_executable(logscan src/main.c)
target_link_libraries(logscan PRIVATE logscan_core)
if(MSVC)
  target_compile_definitions(logscan_core PUBLIC _CRT_SECURE_NO_WARNINGS)
  target_compile_options(logscan_core PRIVATE /W4 /WX)
  target_compile_options(logscan PRIVATE /W4 /WX)
else()
  target_compile_options(logscan_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(logscan PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(LOGSCAN_SANITIZE)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Sanitizers require Clang or GCC")
  endif()
  target_compile_options(logscan_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(logscan_core PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/test_core.c)
  target_link_libraries(core_tests PRIVATE logscan_core)
  add_test(NAME core COMMAND core_tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:logscan>)
endif()
